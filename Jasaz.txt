const XOR_KEY = 0x4f;

function encodeStringToTable(str, key) {
  const utf8Str = unescape(encodeURIComponent(str));
  let out = "";
  for (let i = 0; i < utf8Str.length; i++) {
    out += String.fromCharCode(utf8Str.charCodeAt(i) ^ (key ^ (i % 11)));
  }
  return btoa(out);
}

export function obfuscateWorkerCode(code) {
  const stringTable = [];
  const stringIndexMap = new Map();

  function getOrAddString(strContent) {
    if (stringIndexMap.has(strContent)) return stringIndexMap.get(strContent);
    const idx = stringTable.length;
    stringTable.push(encodeStringToTable(strContent, XOR_KEY));
    stringIndexMap.set(strContent, idx);
    return idx;
  }

  let i = 0;
  const len = code.length;
  let output = "";

  while (i < len) {
    const char = code[i];
    const nextChar = i + 1 < len ? code[i + 1] : "";

    if (char === "/" && nextChar === "/") {
      const end = code.indexOf("\n", i);
      if (end === -1) { output += code.slice(i); break; }
      output += code.slice(i, end);
      i = end;
      continue;
    }

    if (char === "/" && nextChar === "*") {
      const end = code.indexOf("*/", i + 2);
      if (end === -1) { output += code.slice(i); break; }
      output += code.slice(i, end + 2);
      i = end + 2;
      continue;
    }

    if (char === "`") {
      let end = i + 1;
      while (end < len) {
        if (code[end] === "\\") end += 2;
        else if (code[end] === "`") { end++; break; }
        else end++;
      }
      output += code.slice(i, end);
      i = end;
      continue;
    }

    if (char === "/") {
      const trimmed = output.trimEnd();
      const lastChar = trimmed.slice(-1);
      const isRegex = !lastChar || "(=:[,!&|?;{}~^+*%-".includes(lastChar) || /(?:return|case|delete|throw|typeof)\s*$/.test(trimmed);

      if (isRegex) {
        let end = i + 1;
        let inClass = false;
        while (end < len) {
          if (code[end] === "\\") end += 2;
          else if (code[end] === "[" && !inClass) { inClass = true; end++; }
          else if (code[end] === "]" && inClass) { inClass = false; end++; }
          else if (code[end] === "/" && !inClass) {
            end++;
            while (end < len && /[gimsuy]/.test(code[end])) end++;
            break;
          } else end++;
        }
        output += code.slice(i, end);
        i = end;
        continue;
      }
    }

    if (char === '"' || char === "'") {
      const quote = char;
      const strStart = i;
      let strEnd = i + 1;

      while (strEnd < len) {
        if (code[strEnd] === "\\") strEnd += 2;
        else if (code[strEnd] === quote) { strEnd++; break; }
        else if (code[strEnd] === "\n") break;
        else strEnd++;
      }

      const strLiteral = code.slice(strStart, strEnd);
      let rawVal = null;
      try {
        rawVal = quote === '"'
          ? JSON.parse(strLiteral)
          : JSON.parse(`"${strLiteral.slice(1, -1).replace(/\\'/g, "'").replace(/"/g, '\\"')}"`);
      } catch {
        rawVal = strLiteral.slice(1, -1);
      }

      const prevTrimmed = output.trimEnd();
      const isImportExport = /(?:from|import)\s*$/.test(prevTrimmed);
      const isDirective = rawVal === "use strict";

      if (isImportExport || isDirective || !rawVal || rawVal.length === 0) {
        output += strLiteral;
        i = strEnd;
        continue;
      }

      let k = strEnd;
      while (k < len && (code[k] === " " || code[k] === "\t" || code[k] === "\r" || code[k] === "\n")) k++;
      const isFollowedByColon = (code[k] === ":" && code[k + 1] !== ":");
      const lastToken = prevTrimmed.slice(-1);
      const isTernaryOrCondition = lastToken === "?" || /(?:case)\s*$/.test(prevTrimmed);

      if (isFollowedByColon && !isTernaryOrCondition) {
        output += strLiteral;
      } else {
        const idx = getOrAddString(rawVal);
        output += `func_dec(${idx})`;
      }

      i = strEnd;
      continue;
    }

    output += char;
    i++;
  }

  const runtimeHeader = `// @ts-nocheck
const _0x_key = ${XOR_KEY};
const _0x_tbl = ${JSON.stringify(stringTable)};
const _0x_cache = new Map();
function func_dec(idx) {
  if (_0x_cache.has(idx)) return _0x_cache.get(idx);
  const raw = atob(_0x_tbl[idx]);
  let out = "";
  for (let i = 0; i < raw.length; i++) {
    out += String.fromCharCode(raw.charCodeAt(i) ^ (_0x_key ^ (i % 11)));
  }
  try {
    out = decodeURIComponent(escape(out));
  } catch (_) {}
  _0x_cache.set(idx, out);
  return out;
}
`;
  return runtimeHeader + "\n" + output;
}

export async function deployWorker({ accountId, apiToken, workerName, scriptUrl, d1Name, bindingName, kvName, kvBindingName, enableObfuscation, recreateWorker }) {
  if (!accountId || !apiToken || !workerName || !scriptUrl) {
    throw new Error("فیلدهای اجباری (Account ID، Token، نام ورکر و آدرس سورس) ناقص هستند.");
  }

  const cfHeaders = { Authorization: `Bearer ${apiToken}` };

  const scriptRes = await fetch(scriptUrl, {
    headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) PagesDeployer/1.0" },
  });
  if (!scriptRes.ok) {
    throw new Error(`خطا در دانلود سورس: ${scriptRes.status} ${scriptRes.statusText}`);
  }
  let workerScript = await scriptRes.text();

  if (enableObfuscation) {
    workerScript = obfuscateWorkerCode(workerScript);
  }

  const bindings = [];
  let d1DatabaseId = null;
  let kvNamespaceId = null;

  if (d1Name && bindingName) {
    const listRes = await fetch(`https://api.cloudflare.com/client/v4/accounts/${accountId}/d1/database`, {
      headers: cfHeaders,
    });
    const listData = await listRes.json();

    if (!listData.success) {
      const isAuthErr = listData.errors?.some(e => e.code === 10000);
      if (isAuthErr) {
        throw new Error("خطای دسترسی کلودفلر (کد ۱۰۰۰۰): توکن دسترسی 'D1: Edit' را ندارد یا Account ID اشتباه است.");
      }
      throw new Error(`خطای دیتابیس D1: ${JSON.stringify(listData.errors)}`);
    }

    const existingDb = listData.result?.find((db) => db.name === d1Name);

    if (existingDb) {
      d1DatabaseId = existingDb.uuid;
    } else {
      const d1Res = await fetch(`https://api.cloudflare.com/client/v4/accounts/${accountId}/d1/database`, {
        method: "POST",
        headers: { ...cfHeaders, "Content-Type": "application/json" },
        body: JSON.stringify({ name: d1Name }),
      });
      const d1Data = await d1Res.json();
      if (!d1Data.success) {
        throw new Error(`خطا در ایجاد D1: ${JSON.stringify(d1Data.errors)}`);
      }
      d1DatabaseId = d1Data.result.uuid;
    }

    bindings.push({ type: "d1", name: bindingName, id: d1DatabaseId });
  }

  if (kvName && kvBindingName) {
    const kvListRes = await fetch(`https://api.cloudflare.com/client/v4/accounts/${accountId}/storage/kv/namespaces`, {
      headers: cfHeaders,
    });
    const kvListData = await kvListRes.json();

    if (!kvListData.success) {
      throw new Error(`خطا در دسترسی KV: ${JSON.stringify(kvListData.errors)}`);
    }

    const existingKv = kvListData.result?.find((ns) => ns.title === kvName);

    if (existingKv) {
      kvNamespaceId = existingKv.id;
    } else {
      const createKvRes = await fetch(`https://api.cloudflare.com/client/v4/accounts/${accountId}/storage/kv/namespaces`, {
        method: "POST",
        headers: { ...cfHeaders, "Content-Type": "application/json" },
        body: JSON.stringify({ title: kvName }),
      });
      const createKvData = await createKvRes.json();
      if (!createKvData.success) {
        throw new Error(`خطا در ایجاد KV Namespace: ${JSON.stringify(createKvData.errors)}`);
      }
      kvNamespaceId = createKvData.result.id;
    }

    bindings.push({ type: "kv_namespace", name: kvBindingName, namespace_id: kvNamespaceId });
  }

  if (recreateWorker) {
    try {
      await fetch(`https://api.cloudflare.com/client/v4/accounts/${accountId}/workers/scripts/${workerName}`, {
        method: "DELETE",
        headers: cfHeaders,
      });
    } catch (_) {}
  }

  const isModule = workerScript.includes("export default") || workerScript.includes("export ");
  const formData = new FormData();
  const metadata = {
    ...(isModule ? { main_module: "worker.js" } : { body_part: "worker.js" }),
    bindings: bindings,
    compatibility_date: new Date().toISOString().split("T")[0],
  };

  formData.append("metadata", new Blob([JSON.stringify(metadata)], { type: "application/json" }));
  formData.append("worker.js", new Blob([workerScript], { type: "application/javascript+module" }), "worker.js");

  const uploadRes = await fetch(`https://api.cloudflare.com/client/v4/accounts/${accountId}/workers/scripts/${workerName}`, {
    method: "PUT",
    headers: cfHeaders,
    body: formData,
  });

  const uploadData = await uploadRes.json();
  if (!uploadData.success) {
    throw new Error(`خطا از سوی کلودفلر: ${JSON.stringify(uploadData.errors)}`);
  }
  
  await fetch(`https://api.cloudflare.com/client/v4/accounts/${accountId}/workers/scripts/${workerName}/subdomain`, {
    method: "POST",
    headers: { ...cfHeaders, "Content-Type": "application/json" },
    body: JSON.stringify({ enabled: true }),
  });

  let workerUrl = null;
  try {
    const subRes = await fetch(`https://api.cloudflare.com/client/v4/accounts/${accountId}/workers/subdomain`, {
      headers: cfHeaders,
    });
    const subData = await subRes.json();
    if (subData.success && subData.result?.subdomain) {
      workerUrl = `https://${workerName}.${subData.result.subdomain}.workers.dev`;
    }
  } catch (_) {}

  return {
    success: true,
    workerName,
    workerUrl,
    d1DatabaseId,
    kvNamespaceId,
    obfuscated: !!enableObfuscation,
    recreated: !!recreateWorker,
  };
}

export async function onRequestGet({ request }) {

  }

  try {
    const res = await fetch(url, {
      headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) PagesDeployer/1.0" },
    });
    const text = await res.text();
    return new Response(text, {
      headers: {
        "Content-Type": "text/plain; charset=utf-8",

const ORDER_KEYS = ['cs', 'path', 'security', 'alpn', 'encryption', 'fm', 'insecure', 'host', 'fp', 'ech', 'type', 'allowInsecure', 'sni'];

function b64decode(str) {
  try {
    const clean = str.trim().replace(/\s+/g, '').replace(/-/g, '+').replace(/_/g, '/');
    return decodeURIComponent(escape(atob(clean)));
  } catch (_) {
    try {
      return atob(str.trim());
    } catch {
      return '';
    }
  }
}

function b64encode(str) {
  return btoa(unescape(encodeURIComponent(str)));
}

function parseQuery(q) {
  const out = [];
  if (!q) return out;
  q.split('&').forEach((p) => {
    if (!p) return;
    const i = p.indexOf('=');
    const k = i < 0 ? p : p.slice(0, i);
    const v = i < 0 ? '' : p.slice(i + 1);
    try {
      out.push({ k: decodeURIComponent(k), v: decodeURIComponent(v) });
    } catch {
      out.push({ k, v });
    }
  });
  return out;
}

function setParam(params, key, val) {
  const lk = key.toLowerCase();
  let found = false;
  for (let i = params.length - 1; i >= 0; i--) {
    if (params[i].k.toLowerCase() === lk) {
      if (found) { params.splice(i, 1); continue; }
      params[i] = { k: key, v: val };
      found = true;
    }
  }
  if (!found) params.push({ k: key, v: val });
}

function buildQS(params) {
  const idx = {};
  ORDER_KEYS.forEach((k, i) => { idx[k] = i; });
  const a = [], b = [];
  params.forEach((x) => {
    if (idx[x.k.toLowerCase()] !== undefined) a.push(x);
    else b.push(x);
  });
  a.sort((x, y) => idx[x.k.toLowerCase()] - idx[y.k.toLowerCase()]);
  return a.concat(b).map((x) => `${encodeURIComponent(x.k)}=${encodeURIComponent(x.v)}`).join('&');
}

function splitUrl(raw, skip) {
  let b = raw.slice(skip), frag = '';
  const qi = b.indexOf('?');
  const hi = b.indexOf('#', qi >= 0 ? qi : 0);
  if (hi >= 0) {
    frag = b.slice(hi + 1);
    b = b.slice(0, hi);
  }
  let q = '';
  const x = b.indexOf('?');
  if (x >= 0) {
    q = b.slice(x + 1);
    b = b.slice(0, x);
  }
  return { body: b, q, frag };
}

function parseVless(raw) {
  const s = splitUrl(raw, 8);
  const a = s.body.lastIndexOf('@');
  if (a < 0) return null;
  const id = s.body.slice(0, a).trim();
  const hostPort = s.body.slice(a + 1).trim();
  return { proto: 'vless', id, hostPort, p: parseQuery(s.q), f: s.frag };
}

function parseTrojan(raw) {
  const s = splitUrl(raw, 9);
  const a = s.body.lastIndexOf('@');
  if (a < 0) return null;
  const pass = s.body.slice(0, a).trim();
  const hostPort = s.body.slice(a + 1).trim();
  return { proto: 'trojan', pass, hostPort, p: parseQuery(s.q), f: s.frag };
}

function extractIpFromEch(ech) {
  const m = String(ech || '').match(/udp:\/\/([^+\s]+)/i);
  return m ? m[1].replace(/^\[|\]$/g, '') : ech;
}

function extractDomainFromEch(ech) {
  const s = String(ech || '');
  const i = s.indexOf('+');
  return i >= 0 ? s.slice(0, i) : s;
}

function processConfig(raw, opts) {
  let item = null;
  const trimmed = raw.trim();
  if (/^vless:\/\//i.test(trimmed)) item = parseVless(trimmed);
  else if (/^trojan:\/\//i.test(trimmed)) item = parseTrojan(trimmed);
  if (!item) return [];

  const out = [];
  const baseName = item.f ? decodeURIComponent(item.f) : item.proto;

  if (opts.enableEch) {
    opts.echList.forEach((echVal) => {
      const clone = JSON.parse(JSON.stringify(item));
      applyAdvancedParams(clone, opts);

      setParam(clone.p, 'ech', echVal);

      const ip = extractIpFromEch(echVal);
      const domain = extractDomainFromEch(echVal);
      const remark = opts.suffix
        ? opts.suffix.replace(/\{name\}/g, baseName).replace(/\{ip\}/g, ip).replace(/\{domain\}/g, domain).replace(/\{type\}/g, item.proto)
        : `${baseName} ECH ${ip}`;

      const qs = buildQS(clone.p);
      const auth = clone.proto === 'vless' ? clone.id : encodeURIComponent(clone.pass);
      out.push(`${clone.proto}://${auth}@${clone.hostPort}${qs ? `?${qs}` : ''}#${encodeURIComponent(remark)}`);
    });
  } else {

    applyAdvancedParams(item, opts);
    const qs = buildQS(item.p);
    const auth = item.proto === 'vless' ? item.id : encodeURIComponent(item.pass);
    out.push(`${item.proto}://${auth}@${item.hostPort}${qs ? `?${qs}` : ''}#${item.f || ''}`);
  }

  return out;
}

function applyAdvancedParams(item, opts) {
  if (opts.customHost) {
    const parts = item.hostPort.split(':');
    const port = parts.length > 1 ? `:${parts[parts.length - 1]}` : '';
    item.hostPort = `${opts.customHost}${port}`;
  }
  if (opts.fp) {
    setParam(item.p, 'fp', opts.fp);
  }

        "Content-Type": "text/plain; charset=utf-8",
        "Subscription-Userinfo": "upload=0; download=0; total=107374182400; expire=0"
      }
    });

  } catch (err) {

name: Deploy Cloudflare Pages & Setup Telegram Bot

on:
  push:
    branches:
      - main
  workflow_dispatch:

jobs:
  deploy:
    runs-on: ubuntu-latest
    permissions:
      contents: read
      deployments: write
    steps:
      - name: Checkout Repository
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Install Wrangler
        run: npm install -g wrangler

      - name: Ensure Pages Project Exists
        env:
          CLOUDFLARE_API_TOKEN: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          CLOUDFLARE_ACCOUNT_ID: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
        run: |
          wrangler pages project create jasaz --production-branch=main || true

      - name: Sync Secrets to Cloudflare Pages
        env:
          CLOUDFLARE_API_TOKEN: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          CLOUDFLARE_ACCOUNT_ID: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
          TELEGRAM_BOT_TOKEN: ${{ secrets.TELEGRAM_BOT_TOKEN }}
          TELEGRAM_SECRET_TOKEN: ${{ secrets.TELEGRAM_SECRET_TOKEN }}
        run: |
          echo "Syncing Secrets..."
          echo "$TELEGRAM_BOT_TOKEN" | wrangler pages secret put BOT_TOKEN --project-name=jasaz
          echo "$TELEGRAM_SECRET_TOKEN" | wrangler pages secret put BOT_SECRET --project-name=jasaz

      - name: Deploy to Cloudflare Pages
        id: pages_deploy
        env:
          CLOUDFLARE_API_TOKEN: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          CLOUDFLARE_ACCOUNT_ID: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
        run: |
          OUTPUT=$(wrangler pages deploy public --project-name=jasaz --commit-dirty=true)
          echo "$OUTPUT"

          PROJECT_NAME="jasaz"
          CANONICAL_URL="https://${PROJECT_NAME}.pages.dev"

          echo "Canonical URL: $CANONICAL_URL"
          echo "pages_url=$CANONICAL_URL" >> $GITHUB_OUTPUT

      - name: Auto-Register Telegram Webhook
        env:
          TELEGRAM_BOT_TOKEN: ${{ secrets.TELEGRAM_BOT_TOKEN }}
          TELEGRAM_SECRET_TOKEN: ${{ secrets.TELEGRAM_SECRET_TOKEN }}
          PAGES_URL: ${{ steps.pages_deploy.outputs.pages_url }}
        run: |
          if [ -z "${TELEGRAM_BOT_TOKEN}" ]; then
            echo "❌ ERROR: TELEGRAM_BOT_TOKEN secret is empty or not set in GitHub Repository Secrets!"
            exit 1
          fi

          CLEAN_TOKEN=$(echo -n "${TELEGRAM_BOT_TOKEN}" | tr -d '[:space:]' | sed -e 's/^bot//')
          CLEAN_SECRET=$(echo -n "${TELEGRAM_SECRET_TOKEN}" | tr -d '[:space:]')
          
          echo "Token format check: Length is ${#CLEAN_TOKEN} characters."

          echo "Testing Bot Token with getMe..."
          ME_RESPONSE=$(curl -s "https://api.telegram.org/bot${CLEAN_TOKEN}/getMe")
          echo "getMe Response: $ME_RESPONSE"

          if ! echo "$ME_RESPONSE" | grep -q '"ok":true'; then
            echo "❌ The provided TELEGRAM_BOT_TOKEN is invalid or revoked by BotFather!"
            exit 1
          fi

          WEBHOOK_URL="${PAGES_URL}/api/telegram-bot"
          echo "Registering Webhook on: $WEBHOOK_URL"

          RESPONSE=$(curl -s -X POST "https://api.telegram.org/bot${CLEAN_TOKEN}/setWebhook" \
            -H "Content-Type: application/json" \
            -d "{
              \"url\": \"${WEBHOOK_URL}\",
              \"secret_token\": \"${CLEAN_SECRET}\",
              \"drop_pending_updates\": false
            }")

          echo "setWebhook Response: $RESPONSE"

          if echo "$RESPONSE" | grep -q '"ok":true'; then
            echo "✅ Telegram Webhook registered successfully!"
          else
            echo "❌ Failed to register Telegram Webhook!"
            exit 1
          fi
